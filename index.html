Use our API to integrate temp email & SMS into your projects

POST /api/temp/email

fetch('https://api.tempmail.com/v1/temp/email', {
  method: 'POST',
  headers: {'Content-Type':'application/json'},
  body: JSON.stringify({username:'custom', domain:'tempmail.com'})
})

GET /api/messages?email={email}

fetch('https://api.tempmail.com/v1/messages?email=user@tempmail.com')

DELETE /api/email/delete

fetch('https://api.tempmail.com/v1/email/delete', {
  method: 'DELETE',
  body: JSON.stringify({email:'user@tempmail.com'})
})
